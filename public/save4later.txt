<script>
async function displayLastFM() {
            try {
                const response = await fetch('https://lastfm-last-played.biancarosa.com.br/bloxy97/latest-song');
                const data = await response.json();
                const track = data.track;

                document.getElementById("lastheard-container").innerHTML = `
                <div id="trackInfo">
                    <span> <a target="_blank" href="${track.url}">${track.name}</a> - ${track.artist['#text']}</span><br>
                    <img src="${track.image[1]['#text']}">
                </div>`;
            } catch (error) {
                console.error("Fetching failed: ", error);
            }
        };

        displayLastFM();
</script>
